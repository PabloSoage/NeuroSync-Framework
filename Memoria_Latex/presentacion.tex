\section{Introducción y estado del arte}

\begin{frame}
\frametitle{Evolución de la interacción humano-computadora}

\begin{itemize}
\item Las interfaces tradicionales requieren actividad física consciente:
      \begin{itemize}
      \item Comandos de texto, interfaces gráficas, reconocimiento de voz
      \item Limitación: necesidad de movimiento muscular voluntario
      \end{itemize}
\vspace{0.5em}      
\item Las \textbf{Interfaces Cerebro-Computadora (BCI)} representan un paradigma revolucionario:
      \begin{itemize}
      \item Comunicación directa cerebro-dispositivo
      \item Sin necesidad de actividad muscular consciente
      \end{itemize}
\vspace{0.5em}
\item Aplicaciones transformadoras:
      \begin{itemize}
      \item Medicina rehabilitadora (lesiones medulares)
      \item Accesibilidad tecnológica
      \item Nuevos paradigmas de interacción
      \end{itemize}
\end{itemize}

\end{frame}

\begin{frame}
\frametitle{Fundamentos de los sistemas BCI}

\begin{itemize}
\item \textbf{Principio básico}: Captura e interpretación de señales cerebrales
\vspace{0.5em}
\item \textbf{Modalidades principales}:
      \begin{itemize}
      \item \textbf{EEG} (Electroencefalografía): No invasiva, excelente resolución temporal
      \item MEG, fNIRS, técnicas invasivas (ECoG, microelectrodos)
      \end{itemize}
\vspace{0.5em}
\item \textbf{EEG como técnica predominante}:
      \begin{itemize}
      \item Resolución temporal de milisegundos
      \item Dispositivos comerciales accesibles (g.tec Unicorn, Emotiv EPOC X)
      \item Naturaleza completamente no invasiva
      \end{itemize}
\vspace{0.5em}
\item \textbf{Desafío crítico}: Sincronización precisa entre señales cerebrales y acciones del usuario
\end{itemize}

\end{frame}

\begin{frame}
\frametitle{El problema de la fragmentación en herramientas BCI}

\begin{block}{Limitaciones del ecosistema actual}
\begin{itemize}
\item \textbf{Fragmentación de herramientas}: Cada aspecto cubierto por herramientas separadas
      \begin{itemize}
      \item BrainFlow: Solo adquisición EEG
      \item Lab Streaming Layer: Sincronización, pero requiere programación extensiva
      \item MNE-Python: Análisis avanzado, no tiempo real
      \end{itemize}
\vspace{0.5em}
\item \textbf{Ausencia de captura sincronizada}: Ninguna solución integra EEG + eventos de usuario
\vspace{0.5em}
\item \textbf{Barreras de acceso}: Herramientas complejas para desarrolladores expertos
\vspace{0.5em}
\item \textbf{Falta de estandarización}: Formatos propietarios dificultan colaboración
\end{itemize}
\end{block}

\vspace{0.5em}
\textbf{Oportunidad}: Desarrollo de una plataforma integrada que unifique estas capacidades

\end{frame}

\section{Tecnologías seleccionadas para el proyecto}

\begin{frame}
\frametitle{Tecnologías seleccionadas para el proyecto}

\begin{columns}
\begin{column}{0.5\textwidth}
\textbf{Lenguajes principales}
\begin{itemize}
\item \textbf{Python}: Ecosistema científico maduro
      \begin{itemize}
      \item NumPy, SciPy, MNE-Python
      \item Integración natural con BrainFlow
      \end{itemize}
\item \textbf{JavaScript ES6+}: Interfaces web interactivas
\end{itemize}

\vspace{0.5em}
\textbf{Frameworks y bibliotecas}
\begin{itemize}
\item \textbf{PyQt5}: Interfaces gráficas nativas
\item \textbf{FastAPI}: APIs REST de alto rendimiento
\item \textbf{Typer + Rich}: CLI moderna y atractiva
\end{itemize}
\end{column}

\begin{column}{0.5\textwidth}
\textbf{Procesamiento científico}
\begin{itemize}
\item \textbf{NumPy/SciPy}: Computación matricial optimizada
\item \textbf{MNE-Python}: Análisis neurocientífico avanzado
\end{itemize}

\vspace{0.5em}
\textbf{Captura de entrada}
\begin{itemize}
\item \textbf{pynput}: Teclado/ratón multiplataforma
\item \textbf{PySDL2}: Controladores de juego (PS4)
\end{itemize}

\vspace{0.5em}
\textbf{Comunicación}
\begin{itemize}
\item \textbf{WebSockets}: Tiempo real web
\item \textbf{Orjson}: Serialización optimizada
\end{itemize}
\end{column}
\end{columns}

\vspace{1em}
\textbf{Dispositivo de validación}: g.tec Unicorn Hybrid Black (8 canales, 250Hz, 24 bits)
