\section{Curvas}

% Diapositiva: Introducción a las curvas
\begin{frame}{Curvas}
  \begin{itemize}
    \item Definición: Una curva es una función $s(t)$ que describe un camino en el espacio.
    \item Conceptos básicos:
      \begin{itemize}
        \item Puntos de control $\mathbf{c}_i$
        \item Parámetro $\mathbf{t}$
        \item Grado $\mathbf{p}$
      \end{itemize}
    \item Opciones planteadas:
        \begin{itemize}
        \item Curvas de Bézier
        \item Curvas B-Spline
      \end{itemize}
  \end{itemize}
\end{frame}

% Diapositiva: Curvas de Bézier – explicación y fórmula
\begin{frame}{Curvas de Bézier: definición}
  \begin{itemize}
    \item Dada una familia de $n+1$ puntos de control $c_0, \dots, c_n$, la curva de Bézier se define como:
      \[
        B(t) = \sum_{i=0}^{n} \binom{n}{i} (1 - t)^{n - i} t^{i} \, c_i,
        \quad t \in [0,1]
      \]
    \item Resultados suaves
    \item Control global: mover un punto de control modifica la curva entera
  \end{itemize}
\end{frame}

% Diapositiva: Curvas B-Spline – explicación y fórmula
\begin{frame}{Curvas B-Spline}
Definición: combinación lineal de funciones base $N_i^p(t)$ sobre un conjunto de nudos:
          $$s(t) = \sum_{i=0}^{n} N_i^p(t) \, \mathbf{c}_i, \quad t \in [0, 1]$$
          donde:
          \begin{align*}
          N_i^0(t) &= \begin{cases} 1 & \text{si } u_i \le t < u_{i+1}, \\
                                    0 & \text{en otro caso}, \end{cases} \\[3mm]
          N_i^p(t) &= \frac{t - u_i}{u_{i+p} - u_i} N_{i}^{p-1}(t)
                    + \frac{u_{i+p+1} - t}{u_{i+p+1} - u_{i+1}} N_{i+1}^{p-1}(t).
          \end{align*}
\end{frame}

\begin{frame}{B-Splines Racionales}
  \begin{itemize}
    \item Extensión de B-Spline: suma ponderada de cocientes de polinomios:

    \ns (t) = \sumat{i}{0}{n} \nc_i \wh{N}_i^p (t) \, , \qquad
      \text{donde} \quad
      \wh{N}_i^p (t) = \dfrac{\omega_i N_i^p (t)}{\sumat{j}{0}{n} \omega_j N_j^p (t)} \, .
      \end{equation*}
    \item NURBS: uso de un vector de nudos no uniforme.

% Diapositiva: Interpolación de NURBS
\begin{frame}{Sistema a resolver}
Dados:
  \begin{itemize}
    \item $t_i \in [0, 1]$ ($i=0, \ldots, n$)
    \item $q_i$, los puntos de interpolación dados
    \item $N_i^p(t)$ las funciones base
  \end{itemize}
  Buscamos los puntos de control $c_i$ tales que:
    \[
    \left\{
    \begin{aligned}
    N_0^p(t_0) \mathbf{c}_0 + \dots + N_n^p(t_0) \mathbf{c}_n &= \mathbf{q}_0 \\
    N_0^p(t_1) \mathbf{c}_0 + \dots + N_n^p(t_1) \mathbf{c}_n &= \mathbf{q}_1 \\
    &\vdots \\
    N_0^p(t_n) \mathbf{c}_0 + \dots + N_n^p(t_n) \mathbf{c}_n &= \mathbf{q}_n
    \end{aligned}
    \right.
    \]
    % \begin{itemize}
    %     \item Para el caso racional, se sustituyen las funciones base por las funciones base ponderadas con pesos
    % \end{itemize}
\end{frame}

\begin{frame}{Comparativa con una función polinómica}

El error de la interpolación realizada respecto a una función conocida, se calcula de la siguiente manera:

\[
\varepsilon = \frac{\left\lVert f(t) - s(t) \right\rVert}{\left\lVert f(t) \right\rVert}
\]

\section{Superficies}

% Diapositiva: salto de curvas a superficies
\begin{frame}{De curvas a superficies B-Spline}

En las curvas:
  \begin{itemize}
    \item Empleabamos un parámetro $t$
    \end{itemize}
    \[
        s(t) = \sum_{i=0}^{n} N_i^p(t) \, c_i,
        \quad t \in [0, 1]
    \]
  
En las superficies:
  \begin{itemize}
    \item Dos parámetros: $u$, $v$
  \end{itemize}
    \[
        \mathbf{S}(u,v) = \sumat{i}{0}{n} \sumat{j}{0}{m} N_i^p(u) \, N_j^q(v) \, \nc_{i,j}, \quad u,v \in [0, 1]
    \]
\end{frame}

% Diapositiva: cálculo de puntos de control
\begin{frame}{Interpolación de superficies}
  \begin{itemize}
    \item Dada una malla de puntos de interpolación $\mathbf{q}_{ij}$ y los grados $p$ y $q$, buscamos $\mathbf{c}_{ij}$
    \item En cada dirección ($u$, $v$):
    \begin{itemize}
      \item Parametrización: asignar valores $u_i$, $v_j$
      \item Evaluar funciones base
      \item Resolver sistemas lineales (como con las curvas)
    \end{itemize}
    \item Resultado: malla de puntos de control $\mathbf{c}_{ij}$
  \end{itemize}
\end{frame}

% Diapositiva: relevancia de la parametrización

% Diapositiva: Interpolación de NURBS
\begin{frame}{Interpolación racional}
  \begin{itemize}
    \item $3n$ incógnitas:
    \begin{itemize}
        \item Vector de coordenas $x$ : $n$ incógnitas
        \item Vector de coordenas $y$ : $n$ incógnitas
        \item Vector de pesos $\omega$ : $n$ incógnitas
    \end{itemize}
    \item Para poder resolver el sistema, de deben fijar $n$ incógnitas
    \item Solución más práctica: fijar los pesos
  \end{itemize}
  Buscamos los puntos de control $c_i$ tales que:
    \[
    \left\{
    \begin{aligned}
    \wh{N}_0^p(t_0) \mathbf{c}_0 + \dots + \wh{N}_n^p(t_0) \mathbf{c}_n &= \mathbf{q}_0 \\
    \wh{N}_0^p(t_1) \mathbf{c}_0 + \dots + \wh{N}_n^p(t_1) \mathbf{c}_n &= \mathbf{q}_1 \\
    &\vdots \\
    \wh{N}_0^p(t_n) \mathbf{c}_0 + \dots + \wh{N}_n^p(t_n) \mathbf{c}_n &= \mathbf{q}_n
    \end{aligned}
    \right.
    \]
    % \begin{itemize}
    %     \item Para el caso racional, se sustituyen las funciones base por las funciones base ponderadas con pesos
    % \end{itemize}
\end{frame}

\begin{frame}{Interpolación racional en la práctica}

\begin{itemize}
  \item Cada punto se representa en coordenadas homogéneas, como $(\omega x, \omega y, \omega)$, con $\omega \neq 0$.
  \item La proyección al espacio cartesiano:
  \[
  (x, y) = \left( \frac{\omega x}{\omega}, \frac{\omega y}{\omega} \right)
  \]
\end{itemize}

\end{frame}

\begin{frame}{Parametrización lineal}
  \begin{itemize}
    \item Parametrización: $u_i = \frac{i}{n}$ (espaciado uniforme)
    \item Vector de nudos: abierto y uniforme
    \item Ventaja: evaluación directa
    \item Problema: resultados pobres
  \end{itemize}

\begin{frame}{Parametrización de cuerda}
  \begin{itemize}
    \item Parametrización proporcional a distancias reales
    \[ u_0 = 0, \quad
    u_i = u_{i-1} + \| \mathbf{q}_{i,j} - \mathbf{q}_{i-1,j} \|, \quad \text{normalizado a } [0,1] \]

    \[ v_0 = 0, \quad
    v_j = v_{j-1} + \| \mathbf{q}_{i,j} - \mathbf{q}_{i,j-1} \|, \quad \text{normalizado a } [0,1] \, .\]
    \item Vector de nudos: abierto y uniforme
    \item Ventaja: mejora con respecto al caso lineal
    \item Problema: evaluación no directa
  \end{itemize}
\end{frame}

\begin{frame}{Parametrización lineal con vector de nodos de medias}
  \begin{itemize}
    \item Parametrización lineal: $u_i = \frac{i}{n}$
    \item Vector de nudos: medias abiertas (promedios de parámetros internos)
    \end{itemize}
    \[
    U = \left[ \underbrace{0, \dots, 0}_{p+1},\; \frac{1}{p} \sum_{k=1}^{p} u_k,\; \dots,\; \frac{1}{p} \sum_{k=M-p-1}^{M-2} u_k,\; \underbrace{1, \dots, 1}_{p+1} \right]
    \]
    \begin{itemize}
    \item Ventajas:
    \begin{itemize}
      \item Evaluación es directa
      \item Buenos resultados
    \end{itemize}
  \end{itemize}
