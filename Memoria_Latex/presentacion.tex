\section{Introducción y estado del arte}

\begin{frame}
\frametitle{Evolución de la interacción humano-computadora}

\begin{itemize}
\item Las interfaces tradicionales requieren actividad física consciente:
      \begin{itemize}
      \item Comandos de texto, interfaces gráficas, reconocimiento de voz
      \item Limitación: necesidad de movimiento muscular voluntario
      \end{itemize}
\vspace{0.5em}      
\item Las \textbf{Interfaces Cerebro-Computadora (BCI)} representan un paradigma revolucionario:
      \begin{itemize}
      \item Comunicación directa cerebro-dispositivo
      \item Sin necesidad de actividad muscular consciente
      \end{itemize}
\vspace{0.5em}
\item Aplicaciones transformadoras:
      \begin{itemize}
      \item Medicina rehabilitadora (lesiones medulares)
      \item Accesibilidad tecnológica
      \item Nuevos paradigmas de interacción
      \end{itemize}
\end{itemize}

\end{frame}

\begin{frame}
\frametitle{Fundamentos de los sistemas BCI}

\begin{itemize}
\item \textbf{Principio básico}: Captura e interpretación de señales cerebrales
\vspace{0.5em}
\item \textbf{Modalidades principales}:
      \begin{itemize}
      \item \textbf{EEG} (Electroencefalografía): No invasiva, excelente resolución temporal
      \item MEG, fNIRS, técnicas invasivas (ECoG, microelectrodos)
      \end{itemize}
\vspace{0.5em}
\item \textbf{EEG como técnica predominante}:
      \begin{itemize}
      \item Resolución temporal de milisegundos
      \item Dispositivos comerciales accesibles (g.tec Unicorn, Emotiv EPOC X)
      \item Naturaleza completamente no invasiva
      \end{itemize}
\vspace{0.5em}
\item \textbf{Desafío crítico}: Sincronización precisa entre señales cerebrales y acciones del usuario
\end{itemize}

\end{frame}

\begin{frame}
\frametitle{El problema de la fragmentación en herramientas BCI}

\begin{block}{Limitaciones del ecosistema actual}
\begin{itemize}
\item \textbf{Fragmentación de herramientas}: Cada aspecto cubierto por herramientas separadas
      \begin{itemize}
      \item BrainFlow: Solo adquisición EEG
      \item Lab Streaming Layer: Sincronización, pero requiere programación extensiva
      \item MNE-Python: Análisis avanzado, no tiempo real
      \end{itemize}
\vspace{0.5em}
\item \textbf{Ausencia de captura sincronizada}: Ninguna solución integra EEG + eventos de usuario
\vspace{0.5em}
\item \textbf{Barreras de acceso}: Herramientas complejas para desarrolladores expertos
\vspace{0.5em}
\item \textbf{Falta de estandarización}: Formatos propietarios dificultan colaboración
\end{itemize}
\end{block}

\vspace{0.5em}
\textbf{Oportunidad}: Desarrollo de una plataforma integrada que unifique estas capacidades

\end{frame}

\begin{frame}
\frametitle{Tecnologías seleccionadas para el proyecto}

\begin{columns}

\section{Metodología y planificación}

\begin{frame}
\frametitle{Enfoque metodológico adoptado}

\textbf{Modelo incremental-iterativo} para abordar la complejidad de sistemas BCI experimentales.

\vspace{1em}

\begin{columns}
\begin{column}{0.5\textwidth}
\textbf{Incrementalidad}
\begin{itemize}
\item Módulos independientes funcionales
\item Construcción progresiva del sistema
\item Estabilidad garantizada en cada fase
\end{itemize}
\end{column}

\begin{column}{0.5\textwidth}
\textbf{Iteración}
\begin{itemize}
\item Refinamiento continuo
\item Validación temprana de componentes
\item Adaptación a desafíos emergentes
\end{itemize}
\end{column}
\end{columns}

\vspace{1em}

\textbf{Estructura de cada iteración}:
Análisis → Diseño → Implementación → Pruebas → Evaluación

\vspace{0.5em}

\textbf{Validación continua}: Dispositivos reales (g.tec Unicorn) y sintéticos (BrainFlow)

\end{frame}

\begin{frame}
\frametitle{Planificación inicial: enfoque optimista}

\textbf{3 iteraciones, 16 semanas totales}

\begin{itemize}
\item \textbf{Iteración 1}: Desarrollo de módulos independientes (inputs y EEG)
\item \textbf{Iteración 2}: Integración directa mediante Lab Streaming Layer (LSL)
\item \textbf{Iteración 3}: Funcionalidades avanzadas y documentación
\end{itemize}

\vspace{1em}

\textbf{Premisas iniciales}:
\begin{itemize}
\item LSL suficiente para integración
\item Desarrollo lineal sin refactorizaciones mayores
\item Complejidad técnica limitada y predecible
\end{itemize}

\vspace{0.5em}

\begin{alertblock}{Subestimación crítica}
Se asignaron solo 2 semanas para la integración, considerándola trivial
\end{alertblock}

\end{frame}

\begin{frame}
\frametitle{Realidad del desarrollo: múltiples obstáculos}

\textbf{22 semanas reales (+37.5\% de desvío)}

\begin{itemize}
\item \textbf{Iter. 1}: Módulos desarrollados → \textcolor{red}{\textbf{Primer fallo de integración}}
\item \textbf{Iter. 2}: Refactorización SOLID → \textcolor{red}{\textbf{Segundo fallo de integración}}
\item \textbf{Iter. 3}: \textcolor{green}{\textbf{Cambio paradigmático hacia arquitectura API}}
\end{itemize}

\vspace{1em}

\textbf{Principales obstáculos encontrados}:
\begin{itemize}
\item LSL insuficiente para comunicación compleja entre módulos
\item Necesidad de refactorización arquitectónica (principios SOLID)
\item Desarrollo de solución API no planificada inicialmente
\item Pausa no prevista por compromisos académicos (+13 días)
\end{itemize}

\end{frame}
